#large sample 95% simultaneous confidence intervals 
p <- 2

#Using alpha = 5%
alpha <- 0.05
chisq <- qchisq(1- alpha, p) #chi-squared stat 

#Extract means
x1 <- xbar[1]
x2 <- xbar[4]
xbar12 <- matrix(c(x1, x2), nrow = 2)

#Extract variance and covariance 
S11 <- S[1,1]
S12 <- S[1,4]
S22 <- S[4,4]
S12 <- matrix(c(S11, S12, S12, S22), nrow = 2)

#S inverse
S_inv <- solve(S12)

#Eigen values 
eigen <- eigen(S12)




#Axis lengths
(axis1length <- sqrt(eigen$values[1])*sqrt(chisq/n))
[1] 19.05194
(axis2length <- sqrt(eigen$values[2])*sqrt(chisq/n))
[1] 2.123775
#Axis ratios
(axisratio <- sqrt(eigen$values[1])/sqrt(eigen$values[2]))
[1] 8.970791
#Draw the ellipse, same as in textbook
par(mfrow=c(1,1))
phi <- atan2(eigen$vectors[2, 1], eigen$vectors[1, 1]) # angle of major axis
t <- seq(0, 2*pi, 0.01) 
xx <- xbar12[1] + axis1length*cos(t)*cos(phi) - axis2length*sin(t)*sin(phi)
yy <- xbar12[2] + axis1length*cos(t)*sin(phi) + axis2length*sin(t)*cos(phi)
plot(xx, yy, type = 'l', lwd = 2.5, xlab = '', ylab = '') ; title(xlab = bquote(bar(x)[1]), line = 0.2) ; title(ylab = bquote(bar(x)[2]), line = 0) ; lines(c(xbar12[1], xbar12[1]), c(0, xbar12[2]), lty = 5) ; lines(c(0, xbar12[1]), c(xbar12[2], xbar12[2]), lty = 5)


mjrx <- xbar12[1] + c(-1, 1)*axis1length*cos(phi)
mjry <- xbar12[2] + c(-1, 1)*axis1length*sin(phi)
mnrx <- xbar12[1] + c(-1, 1)*axis2length*cos(phi + pi/2) #axes are always orthogonal!

lines(mjrx, mjry, lwd = 2)
