    X = as.matrix(lda_dat[,-1])
lda_dat$group = as.factor(lda_dat$group)
g = length(unique(lda_dat$group))
n_i = table(lda_dat$group)
W = matrix(0, ncol(X), ncol(X))
B = matrix(0, ncol(X), ncol(X))

overall_means = colMeans(X)
group_means = lda_dat %>% 
  group_by(group) %>% 
  summarise(across(everything(), mean), .groups = "drop") %>% 
  dplyr::select(-group) %>% 
  as.matrix()

for(i in 1:g){
  diff = group_means[i,] - overall_means
  B = B + n_i[i] * (diff %*% t(diff))

  Xi = X[as.integer(lda_dat$group) == i,]
  W = W + cov(Xi) * (nrow(Xi) - 1)
}

eig_lda = eigen(solve(W) %*% B)
vec = eig_lda$vectors %>% Re()
Spooled = W / (sum(n_i) - g)
